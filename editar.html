@app.route("/editar/", methods=["GET","POST"]) @login_required def editar(id): conn = get_db() c = conn.cursor() # 🔍 busca registro c.execute("SELECT * FROM registros WHERE id=?", (id,)) r = c.fetchone() if not r: conn.close() return "Registro não encontrado" # 🔒 PERMISSÃO if not is_admin() and r["colaborador"] != current_user.id: conn.close() return "Sem permissão" # ✏️ salvar edição if request.method == "POST": data = request.form["data"] titulo = request.form["titulo"] cliente = request.form["cliente"] nivel = request.form["nivel"] # 🔐 operador não pode trocar o nome if is_admin(): colaborador = request.form["colaborador"] else: colaborador = current_user.id c.execute(""" UPDATE registros SET data=?, titulo=?, cliente=?, colaborador=?, nivel=? WHERE id=? """, (data, titulo, cliente, colaborador, nivel, id)) conn.commit() conn.close() return redirect("/") # 📋 carregar clientes c.execute("SELECT nome FROM clientes") clientes = [x["nome"] for x in c.fetchall()] conn.close() return render_template( "editar.html", r=r, clientes=clientes, is_admin=is_admin() )
